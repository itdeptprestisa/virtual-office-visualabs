# Mode Tampilan 2D Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Menambah mode tampilan 2D (Canvas 2D, gaya ilustrasi ala ZEP, palet terang) yang setara fitur dengan mode 3D dan menjadi bawaan, tanpa mengubah perilaku mode 3D.

**Architecture:** Logika permainan avatar dipindah dari `Player.tsx` ke `controller.ts` (tanpa React/Three) yang dipakai dua "mesin gerak": Rapier (3D, seperti sekarang) dan tabrakan kotak-lingkaran 2D (`motion2d.ts`). Renderer 2D (`world2d/`) menggambar lapisan diam yang di-cache + lapisan dinamis per frame + overlay DOM yang sama dengan 3D. Mode 3D (Three.js/Rapier) dimuat lazy.

**Tech Stack:** TypeScript, React 19, zustand, Canvas 2D, Vite 8 (butuh Node ≥ 22 — pakai `/opt/homebrew/opt/node/bin`), test `node:test` via `tsx`.

**Spec:** `docs/superpowers/specs/2026-10-16-mode-2d-design.md`

## Global Constraints

- **Jangan commit/push** kecuali user memintanya secara eksplisit (aturan proyek ini). Tiap task diakhiri *checkpoint* (test lulus), bukan commit.
- Perilaku mode 3D **tidak berubah** — semua test yang ada (`npm test`, `npm run test:e2e`) tetap lulus.
- Bawaan tampilan **2D**; pilihan disimpan di `localStorage` key `vo-view` (`"2d" | "3d"`), selalu dibungkus try/catch.
- Palet 2D = palet **B (terang)** dari spec (token di Task 4).
- Gambar 2D dibuat lewat kode — **tanpa aset gambar luar**.
- Teks UI baru wajib bahasa Indonesia sebagai kunci `t("…")` + terjemahan di `client/src/i18n.en.ts` (dicek `client/src/i18n.test.ts`).
- Gaya kode: ikuti sekitarnya — komentar bahasa Indonesia, ringkas, tanpa dependensi baru.
- Jalankan Vite/build/e2e dengan `PATH=/opt/homebrew/opt/node/bin:$PATH` (Node 21 bawaan terlalu lama untuk Vite 8).
- Mode 2D **tidak boleh** mengimpor `three`, `@react-three/*`, atau `@react-three/rapier` secara statis.

```ts
// Logika permainan avatar sendiri — tanpa React/Three, dipakai mode 2D & 3D.
// Gerak fisik diserahkan ke "mesin gerak" (Motion): Rapier di 3D, tabrakan kotak-lingkaran di 2D.
import type { Derived, Door, Seat, Zone } from "../../shared/layout";
import { ANIM, EMOTE_KEYS } from "./anims";

export const WALK = 4;
export const RUN = 7;
const SEND_INTERVAL = 1 / 15;
const PROXIMITY_INTERVAL = 1 / 4;

export type Vec3 = [number, number, number];

export interface Motion {
  position(): Vec3;
  setVelocity(vx: number, vz: number): void;
  teleport(x: number, z: number, y?: number): void;
  setEnabled(on: boolean): void; // false selama duduk (tanpa tabrakan)
}

export interface Input {
  held: Set<string>; // tombol yang sedang ditahan
  pressed: Set<string>; // tekanan baru yang belum diproses (C, F, 1/2/3)
  touch: { x: number; y: number }; // joystick sentuh
}

type GameRef = { type: "arcade" | "gameTable"; id: string };

/** Potongan state jaringan yang dibaca controller (subset useNet). */
export interface NetView {
  following: string | null;
  floor: string;
  players: Record<string, { floor: string }>;
  desks: Record<string, { locked: boolean; ownerId: string; ownerName: string }>;
  selfId: string | null;
  zoneId: string | null;
  nearDoor: string | null;
  nearSeat: string | null;
  nearLift: boolean;
  nearGame: GameRef | null;
  sitting?: boolean;
}

export interface ControllerIO {
  state(): NetView;
  patch(p: Partial<NetView> & { zone?: string | null }): void;
  remote(id: string): { p: Vec3; a: number } | undefined;
  remotes(): Iterable<{ p: Vec3; a: number }>;
  world(): Derived;
  zoneAt(x: number, z: number): Zone | null;
  doorNear(x: number, z: number): Door | null;
  seatNear(x: number, z: number, taken: (s: Seat) => boolean): Seat | null;
  ejectPoint(zone: Zone, x: number, z: number): [number, number];
  gameNear(x: number, z: number): GameRef | null;
  openGame(g: GameRef): void;
  sendMove(p: Vec3, r: number, a: number): void;
  sendDoor(id: string, inside: boolean): void;
  updateProximity(x: number, z: number): void;
  teleportNear(x: number, z: number, floor: string): void;
  focusNotice(ownerName: string): void;
  setSelfPos(p: Vec3): void;
}

export interface Controller {
  step(dt: number, now: number, camYaw: number): void;
  readonly anim: number;
  readonly facing: number; // arah hadap (rotation.y), 0 = menghadap +z (selatan)
  readonly seat: Seat | null;
  teleport(x: number, z: number): void;
  toggleSit(): void;
  standUp(): void;
}

export function dampAngle(from: number, to: number, k: number) {
  let d = to - from;
  d = Math.atan2(Math.sin(d), Math.cos(d));
  return from + d * k;
}

export function createController(motion: Motion, input: Input, io: ControllerIO): Controller {
  let anim: number = ANIM.idle;
  let facing = 0;
  let emote: { anim: number; until: number } | null = null;
  let seat: Seat | null = null;
  let sendTimer = 0;
  let proximityTimer = 0;
  let lastSent = "";
  let lastValid: [number, number] | null = null;
  let lastNotice = -Infinity;
  let followStuck = { dist: Infinity, since: 0 };

  /** Kursi sedang diduduki orang lain? (avatar lain berstatus duduk di dekat kursi itu) */
  const seatTaken = (s: Seat) => {
    for (const r of io.remotes()) if (r.a === ANIM.sit && Math.hypot(r.p[0] - s.x, r.p[2] - s.z) < 0.4) return true;
    return false;
  };

  const standUp = () => {
    const s = seat;
    if (!s) return;
    seat = null;
    motion.setEnabled(true);
    motion.teleport(s.stand[0], s.stand[1]);
    motion.setVelocity(0, 0);
    io.patch({ sitting: false });
  };

  const sitOn = (s: Seat) => {
    seat = s;
    emote = null;
    // Selama duduk, badan fisik dinonaktifkan supaya tidak bertabrakan dengan sofa/meja.
    motion.setEnabled(false);
    motion.teleport(s.x, s.z);
    facing = s.yaw;
    io.patch({ sitting: true, nearSeat: null });
  };

  const toggleSit = () => {
    if (seat) return standUp();
    const [x, , z] = motion.position();
    const s = io.seatNear(x, z, seatTaken);
    if (s) sitOn(s);
  };

  const teleport = (x: number, z: number) => {
    if (seat) standUp();
    motion.teleport(x, z);
    motion.setVelocity(0, 0);
  };

  function step(dt: number, now: number, camYaw: number) {
    const k = input.held;
    const pressed = input.pressed;

    // Input relatif terhadap arah kamera: W = maju menjauhi kamera. Joystick sentuh ikut dijumlahkan.
    let fwd = input.touch.y;
    let side = input.touch.x;
    if (k.has("KeyW") || k.has("ArrowUp")) fwd += 1;
    if (k.has("KeyS") || k.has("ArrowDown")) fwd -= 1;
    if (k.has("KeyA") || k.has("ArrowLeft")) side -= 1;
    if (k.has("KeyD") || k.has("ArrowRight")) side += 1;
    const sy = Math.sin(camYaw);
    const cy = Math.cos(camYaw);
    let x = -sy * fwd + cy * side;
    let z = -cy * fwd - sy * side;
    let len = Math.hypot(x, z);
    if (len < 0.15) len = 0; // zona mati joystick
    if (len === 0) x = z = 0;
    let running = k.has("ShiftLeft") || k.has("ShiftRight") || Math.hypot(input.touch.x, input.touch.y) > 0.92;

    // Mode ikuti: jalan otomatis ke orang yang diikuti; tombol gerak = berhenti mengikuti.
    const following = io.state().following;
    if (following && len > 0) io.patch({ following: null });
    else if (following) {
      const target = io.remote(following);
      const info = io.state().players[following];
      const [hx, , hz] = motion.position();
      if (!target || !info) io.patch({ following: null });
      else if (info.floor !== io.state().floor) {
        io.teleportNear(target.p[0], target.p[2], info.floor); // ikut naik lift
      } else {
        const dx = target.p[0] - hx;
        const dz = target.p[2] - hz;
        const dist = Math.hypot(dx, dz);
        if (dist > 1.8) {
          x = dx / dist;
          z = dz / dist;
          len = 1;
          running = dist > 6;
          // Tersangkut (dinding/furnitur) → lompat ke dekatnya.
          if (dist >= followStuck.dist - 0.05) {
            if (now - followStuck.since > 2.5) {
              io.teleportNear(target.p[0], target.p[2], io.state().floor);
              followStuck = { dist: Infinity, since: now };
            }
          } else followStuck = { dist, since: now };
        }
      }
    }
    const speed = running ? RUN : WALK;

    // Duduk / berdiri (C). Bergerak saat duduk = berdiri.
    if (pressed.has("KeyC")) {
      pressed.delete("KeyC");
      toggleSit();
    }
    if (seat && len > 0 && !following) standUp();
    if (seat && following) io.patch({ following: null });

    // Emote (1/2/3) — dibatalkan begitu avatar bergerak.
    for (const code in EMOTE_KEYS) {
      if (!pressed.has(code)) continue;
      pressed.delete(code);
      if (seat) continue;
      const e = EMOTE_KEYS[code];
      emote = { anim: e.anim, until: e.duration === null ? Infinity : now + e.duration };
    }
    if (len > 0 || (emote && now >= emote.until)) emote = null;

    if (seat) {
      anim = ANIM.sit;
    } else {
      anim = len > 0 ? (running ? ANIM.run : ANIM.walk) : (emote?.anim ?? ANIM.idle);
      if (len > 0) {
        const n = Math.max(1, len);
        x /= n;
        z /= n;
        facing = dampAngle(facing, Math.atan2(x, z), 1 - Math.exp(-12 * dt));
      }
      motion.setVelocity(x * speed, z * speed);
    }

    // Meja terkunci milik orang lain tidak bisa dimasuki: kembalikan ke posisi sah terakhir / dorong keluar.
    const p = motion.position();
    let px = p[0];
    let pz = p[2];
    const { desks, selfId } = io.state();
    const lockedFor = (qx: number, qz: number) => {
      const zn = io.zoneAt(qx, qz);
      const dk = zn ? desks[zn.id] : undefined;
      return dk && dk.locked && dk.ownerId !== selfId ? zn : null;
    };
    const blocked = lockedFor(px, pz);
    if (blocked) {
      if (seat) standUp();
      const lv = lastValid;
      const [ex, ez] = lv && !lockedFor(lv[0], lv[1]) ? lv : io.ejectPoint(blocked, px, pz);
      motion.teleport(ex, ez, p[1]);
      motion.setVelocity(0, 0);
      if (now - lastNotice > 3) {
        lastNotice = now;
        io.focusNotice(desks[blocked.id].ownerName);
      }
      px = ex;
      pz = ez;
    } else {
      lastValid = [px, pz];
    }
    io.setSelfPos([px, p[1], pz]);

    const here = io.zoneAt(px, pz);
    if (io.state().zoneId !== (here?.id ?? null)) io.patch({ zone: here?.name ?? null, zoneId: here?.id ?? null });

    // Pintu terdekat: F untuk buka/tutup/ketuk. Tanpa pintu di dekat, F membuka mesin arcade / meja game terdekat.
    const door = io.doorNear(px, pz);
    if (io.state().nearDoor !== (door?.id ?? null)) io.patch({ nearDoor: door?.id ?? null });
    if (pressed.has("KeyF")) {
      pressed.delete("KeyF");
      const game = io.state().nearGame;
      if (door) io.sendDoor(door.id, here?.id === door.room);
      else if (game) io.openGame(game);
    }

    // Audio/video proximity + kursi/lift/game terdekat (beberapa kali per detik saja).
    proximityTimer += dt;
    if (proximityTimer >= PROXIMITY_INTERVAL) {
      proximityTimer = 0;
      io.updateProximity(px, pz);
      const near = seat ? null : (io.seatNear(px, pz, seatTaken)?.id ?? null);
      if (io.state().nearSeat !== near) io.patch({ nearSeat: near });
      const lift = io.world().elevators.some((e) => Math.hypot(e.exit[0] - px, e.exit[1] - pz) < 2.2);
      if (io.state().nearLift !== lift) io.patch({ nearLift: lift });
      const game = io.gameNear(px, pz);
      if (io.state().nearGame?.id !== game?.id) io.patch({ nearGame: game });
    }

    // Kirim posisi maks 15x/detik, hanya kalau berubah.
    sendTimer += dt;
    if (sendTimer >= SEND_INTERVAL || lastSent === "") {
      sendTimer = 0;
      const sp: Vec3 = [px, p[1], pz];
      const key = `${sp.map((n) => n.toFixed(2)).join()}|${facing.toFixed(2)}|${anim}`;
      if (key !== lastSent) {
        lastSent = key;
        io.sendMove(sp, facing, anim);
      }
    }
  }

  return {
    step,
    get anim() {
      return anim;
    },
    get facing() {
      return facing;
    },
    get seat() {
      return seat;
    },
    teleport,
    toggleSit,
    standUp,
  };
}
```

```ts
// daun pintu tertutup, furnitur padat — ukuran sama dengan collider 3D).
import { WORLD, toWorld, type Derived, type Floor } from "../../shared/layout";
import { COLLIDERS } from "../../shared/furniture";
import type { Motion, Vec3 } from "./controller";

export interface Box {
  x0: number;
  x1: number;
  z0: number;
  z1: number;
}

export const RADIUS = 0.3;
const GLASS = 0.05; // setengah tebal dinding kaca / daun pintu
const OUTER = 0.15; // setengah tebal dinding luar
const CELL = 2;
const MAX_STEP = 0.2;
const MAX_DT = 0.1;

const seg = (orient: "x" | "z", a: [number, number], c: number): Box =>
  orient === "x" ? { x0: a[0], x1: a[1], z0: c - GLASS, z1: c + GLASS } : { x0: c - GLASS, x1: c + GLASS, z0: a[0], z1: a[1] };

export function buildObstacles(floor: Floor, d: Derived, doors: Record<string, { open: boolean }>): Box[] {
  const { minX, maxX, minZ, maxZ } = WORLD;
  const boxes: Box[] = [
    { x0: minX - 1, x1: minX + OUTER, z0: minZ - 1, z1: maxZ + 1 },
    { x0: maxX - OUTER, x1: maxX + 1, z0: minZ - 1, z1: maxZ + 1 },
    { x0: minX - 1, x1: maxX + 1, z0: minZ - 1, z1: minZ + OUTER },
    { x0: minX - 1, x1: maxX + 1, z0: maxZ - OUTER, z1: maxZ + 1 },
  ];
  for (const w of d.walls) boxes.push(seg(w.orient, w.a, w.c));
  for (const dr of d.doors) if (doors[dr.id] && !doors[dr.id].open) boxes.push(seg(dr.orient, dr.a, dr.c));
  for (const it of floor.items) {
    const col = COLLIDERS[it.type];
    if (!col) continue;
    const [hx, , hz] = col.half;
    const [px, , pz] = col.pos;
    const a = toWorld(it, px - hx, pz - hz);
    const b = toWorld(it, px + hx, pz + hz);
    boxes.push({ x0: Math.min(a[0], b[0]), x1: Math.max(a[0], b[0]), z0: Math.min(a[1], b[1]), z1: Math.max(a[1], b[1]) });
  }
  return boxes;
}

/** Grid sederhana supaya tiap langkah hanya memeriksa penghalang di sekitar avatar. */
export class Grid {
  private cells = new Map<string, Box[]>();
  constructor(boxes: Box[]) {
    for (const b of boxes)
      for (let i = Math.floor(b.x0 / CELL); i <= Math.floor(b.x1 / CELL); i++)
        for (let j = Math.floor(b.z0 / CELL); j <= Math.floor(b.z1 / CELL); j++) {
          const k = `${i},${j}`;
          const list = this.cells.get(k);
          if (list) list.push(b);
          else this.cells.set(k, [b]);
        }
  }
  query(x0: number, z0: number, x1: number, z1: number): Box[] {
    const out = new Set<Box>();
    for (let i = Math.floor(x0 / CELL); i <= Math.floor(x1 / CELL); i++)
      for (let j = Math.floor(z0 / CELL); j <= Math.floor(z1 / CELL); j++) for (const b of this.cells.get(`${i},${j}`) ?? []) out.add(b);
    return [...out];
  }
}

const hits = (b: Box, x: number, z: number) => {
  const cx = Math.max(b.x0, Math.min(b.x1, x));
  const cz = Math.max(b.z0, Math.min(b.z1, z));
  return (x - cx) ** 2 + (z - cz) ** 2 < RADIUS * RADIUS;
};

/** Langkah dari (ox,oz) ke (nx,nz) terhalang? Penghalang yang sudah ditempati sejak awal diabaikan (supaya bisa keluar). */
function blocked(grid: Grid, ox: number, oz: number, nx: number, nz: number) {
  for (const b of grid.query(nx - RADIUS, nz - RADIUS, nx + RADIUS, nz + RADIUS)) if (hits(b, nx, nz) && !hits(b, ox, oz)) return true;
  return false;
}

/** Gerak dengan tabrakan, per sumbu (x lalu z) supaya avatar meluncur menyusuri dinding. */
export function slide(grid: Grid, x: number, z: number, dx: number, dz: number): [number, number] {
  const steps = Math.max(1, Math.ceil(Math.hypot(dx, dz) / MAX_STEP));
  for (let i = 0; i < steps; i++) {
    const nx = x + dx / steps;
    if (!blocked(grid, x, z, nx, z)) x = nx;
    const nz = z + dz / steps;
    if (!blocked(grid, x, z, x, nz)) z = nz;
  }
  return [x, z];
}

/** Mesin gerak 2D untuk controller. */
export function createMotion2D(getGrid: () => Grid, start: [number, number]): Motion & { integrate(dt: number): void } {
  let x = start[0];
  let z = start[1];
  let vx = 0;
  let vz = 0;
  let enabled = true;
  return {
    position: (): Vec3 => [x, 0, z],
    setVelocity(a, b) {
      vx = a;
      vz = b;
    },
    teleport(nx, nz) {
      x = nx;
      z = nz;
    },
    setEnabled(on) {
      enabled = on;
    },
    integrate(dt) {
      if (!enabled) return;
      const t = Math.min(dt, MAX_DT);
      [x, z] = slide(getGrid(), x, z, vx * t, vz * t);
    },
  };
}
```

```ts
// Posisi orang lain ditampilkan sedikit "di masa lalu" lalu diinterpolasi di antara dua posisi yang sudah diterima,
// sehingga gerak tetap mulus walau paket datang 15x/detik dengan jeda tidak rata. Dipakai 3D & 2D.
import { remoteState, type Vec3 } from "./net";

export const INTERP_DELAY = 120; // ms

const lerpAngle = (a: number, b: number, k: number) => a + Math.atan2(Math.sin(b - a), Math.cos(b - a)) * k;

export function sampleRemote(id: string, nowMs: number): { p: Vec3; r: number; a: number } | null {
  const s = remoteState.get(id);
  if (!s || !s.buf.length) return null;
  const buf = s.buf;
  const rt = nowMs - INTERP_DELAY;
  let i = buf.length - 1;
  while (i > 0 && buf[i].t > rt) i--;
  const a = buf[i];
  const b = buf[i + 1];
  if (b && rt >= a.t) {
    const k = Math.min(1, (rt - a.t) / Math.max(1, b.t - a.t));
    return { p: [a.p[0] + (b.p[0] - a.p[0]) * k, a.p[1] + (b.p[1] - a.p[1]) * k, a.p[2] + (b.p[2] - a.p[2]) * k], r: lerpAngle(a.r, b.r, k), a: s.a };
  }
  // Belum ada data sebelum rt (baru muncul) atau sudah melewati data terakhir (berhenti): tahan di titik itu.
  const at = rt < a.t ? a : buf[buf.length - 1];
  return { p: [at.p[0], at.p[1], at.p[2]], r: at.r, a: s.a };
}
```

```ts
export function shadowed(g: CanvasRenderingContext2D, fn: () => void) {
  g.save();
  const s = g.getTransform?.().a || 48;
  g.shadowColor = "rgba(0,0,0,.2)";
  g.shadowBlur = 0.12 * s;
  g.shadowOffsetY = 0.06 * s;
  fn();
  g.restore();
}
export function shade(hex: string, k: number) {
  const n = parseInt(hex.slice(1), 16);
  const f = (v: number) => Math.max(0, Math.min(255, Math.round(v * k)));
  return `rgb(${f(n >> 16)},${f((n >> 8) & 255)},${f(n & 255)})`;
}
```

```ts
// Kamera 2D: pusat (x,z) dunia, zoom, ukuran kanvas (px CSS). Sumbu y layar = z dunia.
export interface Cam {
  x: number;
  z: number;
  zoom: number;
  w: number;
  h: number;
}
export const BASE_PX = 48; // piksel per meter pada zoom 1
export const ZOOM_MIN = 0.5;
export const ZOOM_MAX = 2.5;
export const ppm = (c: Cam) => BASE_PX * c.zoom;
export const toScreen = (c: Cam, x: number, z: number): [number, number] => [(x - c.x) * ppm(c) + c.w / 2, (z - c.z) * ppm(c) + c.h / 2];
export const toWorld2 = (c: Cam, sx: number, sy: number): [number, number] => [(sx - c.w / 2) / ppm(c) + c.x, (sy - c.h / 2) / ppm(c) + c.z];
export function viewRect(c: Cam, margin = 0) {
  const hw = c.w / 2 / ppm(c) + margin;
  const hh = c.h / 2 / ppm(c) + margin;
  return { x0: c.x - hw, z0: c.z - hh, x1: c.x + hw, z1: c.z + hh };
}
```

```ts
function grid(g: CanvasRenderingContext2D, x0: number, z0: number, x1: number, z1: number, step: number, col: string, lw: number) {
```

```ts
// Lapisan dinamis per frame, digambar di atas lapisan diam dengan transform meter → layar.
import { PROXIMITY, type Derived, type Floor, type Zone } from "../../../shared/layout";
import type { AvatarConfig } from "../avatar/config";
import { drawAvatar } from "./draw/avatar";
import { DRAW, TALL } from "./draw/furniture";
import { ppm, type Cam } from "./camera";

export interface Body2D {
  id: string;
  cfg: AvatarConfig;
  x: number;
  z: number;
  facing: number;
  anim: number;
  hand: boolean;
}

export interface DynFrame {
  floor: Floor;
  derived: Derived;
  zone: Zone | null; // zona aktif diri sendiri
  self: Body2D;
  others: Body2D[];
  ring: { show: boolean; someone: boolean };
  t: number; // detik
}

export function drawDynamic(g: CanvasRenderingContext2D, cam: Cam, f: DynFrame) {
  const p = ppm(cam);
  g.save();
  g.setTransform(p * devicePixelRatio, 0, 0, p * devicePixelRatio, (cam.w / 2 - cam.x * p) * devicePixelRatio, (cam.h / 2 - cam.z * p) * devicePixelRatio);
  // sorotan zona aktif
  const zn = f.zone;
  if (zn) {
    g.fillStyle = zn.color + "33";
    g.strokeStyle = zn.color;
    g.lineWidth = 0.06;
    g.beginPath();
    if (zn.shape === "rect") g.rect(zn.x[0], zn.z[0], zn.x[1] - zn.x[0], zn.z[1] - zn.z[0]);
    else g.arc(zn.cx, zn.cz, zn.r, 0, Math.PI * 2);
    g.fill();
    g.stroke();
  }
  // lingkaran jarak dengar
  if (f.ring.show) {
    const col = f.ring.someone ? "#4caf72" : "#5b8def";
    g.fillStyle = col + (f.ring.someone ? "1a" : "0d");
    g.strokeStyle = col;
    g.lineWidth = 0.06;
    g.beginPath();
    g.arc(f.self.x, f.self.z, PROXIMITY.hear, 0, Math.PI * 2);
    g.fill();
    g.stroke();
  }
  // avatar, urut dari utara ke selatan supaya yang di depan menutupi
  const bodies = [...f.others, f.self].sort((a, b) => a.z - b.z);
  for (const b of bodies) drawAvatar(g, b.cfg, b.x, b.z, b.facing, b.anim, f.t, b.hand);
  // atas tanaman di depan avatar yang berdiri tepat di belakangnya
  for (const it of f.floor.items) {
    if (!TALL.has(it.type)) continue;
    if (!bodies.some((b) => Math.abs(b.x - it.x) < 0.6 && b.z < it.z && it.z - b.z < 0.8)) continue;
    g.save();
    g.translate(it.x, it.z);
    DRAW[it.type](g, it);
    g.restore();
  }
  g.restore();
}
```

```tsx
import { useMedia } from "../media";

/** Mode tampilan 2D: kanvas + loop frame; logika permainan memakai controller yang sama dengan 3D. */
export function World2D() {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const labelsRef = useRef<HTMLDivElement>(null);
  const self = useNet((s) => s.self)!;
  const players = useNet((s) => s.players);
  const floorId = useNet((s) => s.floor);
  const { held, pressed } = useKeyboard();

  const gridRef = useRef<Grid | null>(null);
  const motion = useMemo(() => {
    const s = useNet.getState().selfPos?.p ?? useNet.getState().spawn;
    return createMotion2D(() => gridRef.current ?? new Grid([]), [s[0], s[2]]);
  }, []);
  const controller = useMemo(() => createController(motion, { held: held.current, pressed: pressed.current, touch: touchInput }, defaultIO), [motion, held, pressed]);
  useEffect(() => setActiveController(controller), [controller]);

  useEffect(() => {
    const rebuild = () => {
      const { layout, floor, derived, doors } = useNet.getState();
      const f = layout?.floors.find((x) => x.id === floor);
      gridRef.current = f && derived ? new Grid(buildObstacles(f, derived, doors)) : null;
    };
    rebuild();
    return useNet.subscribe((s, prev) => {
      if (s.layout !== prev.layout || s.floor !== prev.floor || s.derived !== prev.derived || s.doors !== prev.doors) rebuild();
    });
  }, []);

  // Kursi yang diduduki hilang (denah diubah) → berdiri.
  const derived = useNet((s) => s.derived);
  useEffect(() => {
    const s = controller.seat;
    if (s && !currentWorld().seats.some((x) => x.id === s.id)) controller.standUp();
  }, [derived, controller]);

  const cam = useRef<Cam>({ x: motion.position()[0], z: motion.position()[2], zoom: 1, w: 1, h: 1 });
  const layer = useMemo(() => new StaticLayer(), []);

  useEffect(() => {
    const canvas = canvasRef.current!;
    const g = canvas.getContext("2d")!;
    let raf = 0;
    let last = performance.now();
    const resize = () => {
      const r = canvas.getBoundingClientRect();
      cam.current.w = r.width;
      cam.current.h = r.height;
      canvas.width = Math.round(r.width * devicePixelRatio);
      canvas.height = Math.round(r.height * devicePixelRatio);
    };
    resize();
    const ro = new ResizeObserver(resize);
    ro.observe(canvas);
    const onWheel = (e: WheelEvent) => {
      e.preventDefault();
      cam.current.zoom = Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, cam.current.zoom * Math.exp(-e.deltaY * 0.0015)));
    };
    canvas.addEventListener("wheel", onWheel, { passive: false });

    const frame = (now: number) => {
      raf = requestAnimationFrame(frame);
      const dt = Math.min(0.1, (now - last) / 1000);
      last = now;
      controller.step(dt, now / 1000, 0); // 2D: "maju" = utara (atas layar)
      motion.integrate(dt);
      const [px, , pz] = motion.position();
      const c = cam.current;
      const k = 1 - Math.exp(-8 * dt);
      c.x += (px - c.x) * k;
      c.z += (pz - c.z) * k;

      const f = st.layout?.floors.find((x) => x.id === st.floor);

      g.clearRect(0, 0, c.w, c.h);
      layer.draw(g, c, { floor: f, derived: st.derived });
      const hands = new Set(st.directory.filter((u) => u.hand).map((u) => u.id));
      const selfBody: Body2D = { id: self.id, cfg: st.self!.avatar, x: px, z: pz, facing: controller.facing, anim: controller.anim, hand: hands.has(self.id) };
      const others: Body2D[] = [];
      for (const pl of Object.values(st.players) as PlayerInfo[]) {
        if (pl.floor !== st.floor) continue;
        const s = sampleRemote(pl.id, now);
        if (s) others.push({ id: pl.id, cfg: pl.avatar, x: s.p[0], z: s.p[2], facing: s.r, anim: s.a, hand: hands.has(pl.id) });
      }
      const zone = st.zoneId ? (st.derived.zones.find((z) => z.id === st.zoneId) ?? null) : null;

          if (!el) continue;

          el.style.transform = `translate(${sx}px, ${sy}px) translate(-50%, -100%)`;
        }
    };
    raf = requestAnimationFrame(frame);
    return () => {
      cancelAnimationFrame(raf);
      ro.disconnect();
      canvas.removeEventListener("wheel", onWheel);
    };

    <div className="world2d">
      <canvas ref={canvasRef} />
      <div ref={labelsRef} className="overlay2d">

          <AvatarLabel id={self.id} name={self.name} color={self.avatar.topColor} isSelf />

            <AvatarLabel id={p.id} name={p.name} color={p.avatar.topColor} />
          </div>
        ))}
      </div>
    </div>
  );
}
```

```ts
import { FOOTPRINT } from "../../../shared/furniture";

/** Furnitur di titik (x,z): yang radius sorotannya memuat titik, dipilih yang terkecil (paling spesifik). */
export function hitItem(items: Item[], x: number, z: number): Item | null {
  let best: Item | null = null;
  for (const it of items) {
    const r = FOOTPRINT[it.type];
    if (Math.hypot(it.x - x, it.z - z) <= r && (!best || r < FOOTPRINT[best.type])) best = it;
  }
  return best;
}
```

```tsx
/** Pratinjau avatar 2D: berputar pelan melewati 4 arah hadap. */
export function Preview2D({ config }: { config: AvatarConfig }) {
  const ref = useRef<HTMLCanvasElement>(null);
  useEffect(() => {
    const c = ref.current!;
    const g = c.getContext("2d")!;
    let raf = 0;
    const frame = (now: number) => {
      raf = requestAnimationFrame(frame);
      const r = c.getBoundingClientRect();
      c.width = r.width * devicePixelRatio;
      c.height = r.height * devicePixelRatio;
      const s = (c.height / 1.6);
      g.setTransform(s, 0, 0, s, c.width / 2, c.height * 0.85);
      g.clearRect(-10, -10, 20, 20);
      const facing = [0, Math.PI / 2, Math.PI, -Math.PI / 2][Math.floor(now / 1500) % 4];
      drawAvatar(g, config, 0, 0, facing, ANIM.idle, now / 1000, false);
    };
    raf = requestAnimationFrame(frame);
    return () => cancelAnimationFrame(raf);
  }, [config]);
  return <canvas ref={ref} className="avatar-preview2d" style={{ width: "100%", height: "100%" }} />;
}
```
